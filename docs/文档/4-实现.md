### 3.1 ShardingSphere 分片

```yaml
# sharding.yaml
dataSources:
  ds_0: {jdbcUrl: jdbc:mysql://localhost:3306/db_0, ...}
  ds_1: {jdbcUrl: jdbc:mysql://localhost:3307/db_1, ...}
  ds_2: {jdbcUrl: jdbc:mysql://localhost:3308/db_2, ...}

rules:
  - !SHARDING
    tables:
      user:
        actualDataNodes: ds_${0..2}.user_${0..5}
        databaseStrategy: {shardingColumn: phone, algorithm: database_mod}
        tableStrategy:    {shardingColumn: phone, algorithm: table_mod}
  - !SINGLE
    tables: ["*.*"]
    defaultDataSource: ds_0
```

## 4. 核心代码片段

### 4.1 位图空闲判断

```java
private boolean isSeatFreeInMemory(byte[] bitmapBytes, long seatStartBit,
                                   int userStartSection, int userEndSection) {
    long rangeStartBit = seatStartBit + userStartSection - 1;
    long rangeEndBit   = seatStartBit + userEndSection - 1;
    int startByte = (int) (rangeStartBit / 8);
    int endByte   = (int) (rangeEndBit / 8);

    // 一次读取拼接成 int
    int value = 0;
    for (int i = 0; i <= endByte - startByte; i++) {
        value |= (bitmapBytes[startByte + i] & 0xFF) << (i * 8);
    }

    // 构建 mask 并一次 AND 判断
    int rangeBits = userEndSection - userStartSection + 1;
    int startBitInValue = (int) (rangeStartBit % 8);
    int mask = ((1 << rangeBits) - 1) << startBitInValue;

    return (value & mask) == 0;
}
```

### 4.2 Token 桶限流（Lua）

```lua
-- token_bucket.lua
local key = KEYS[1]
local requested = tonumber(ARGV[1])
local current = tonumber(redis.call('GET', key) or '0')
if current >= requested then
    redis.call('DECRBY', key, requested)
    return 1  -- 放行
end
return -1  -- 拒绝
```

## 5. 启动步骤

```bash
# 1. 编译
mvn clean package -DskipTests

# 2. 启动服务（按依赖顺序）
mvn -pl user-service spring-boot:run &
mvn -pl ticket-service spring-boot:run &
mvn -pl order-service spring-boot:run &
mvn -pl gateway spring-boot:run &

# 3. 启动前端
cd frontend && npm install && npm run dev
# 访问 http://localhost:5173
```
